Compile Kernel Module

ใช้คอนฟิกเดียวกับ Kernel ที่จะโหลดโมดูลเข้าไป ใช้ Makefile และ Kconfig

ตัวอย่างที่ 1 Makefile

KDIR = /lib/modules/`uname -r`/build

kbuild:
        make -C $(KDIR) M=`pwd`

clean:
        make -C $(KDIR) M=`pwd` clean

Kconfig

EXTRA_CFLAGS = -Wall -g

obj-m        = modul.o
